Tiled Convolution Kernel
const T = 16; const S = T + 2R  # tile 16, loaded 20
function convolve_tiled!(out, F, w)
tile = CuStaticSharedArray(Float32, (S, S))
tr, tc = threadIdx().x, threadIdx().y  # 1..16
r0 = (blockIdx().x - Int32(1)) * T  # tile origin in the frame
c0 = (blockIdx().y - Int32(1)) * T
H, W = size(F)
for i in tr:T:S, j in tc:T:S  # 1. load together
@inbounds tile[i, j] = F[clamp(r0 + i - R, 1, H), clamp(c0 + j - R, 1, W)]
end
sync_threads()  # 2. synchronise
r, c = r0 + tr, c0 + tc  # 3. compute from the tile
if r <= H && c <= W
acc = 0f0
for j in -R:R, i in -R:R
@inbounds acc += w[i + R + 1, j + R + 1] * tile[tr + R + i, tc + R + j]
end
@inbounds out[r, c] = acc
end
return nothing
end

Intensity histogram: one frame in 256 numbers
Normal view  Covered lens  Blinded by light

Tampering indicators: a covered lens collapses the histogram into the dark bins, a torch

Detection: the distance between this frame’s 256 bins and their running average, one
number per frame; a jump marks possible tampering or a change of lighting

Type, memory and separability of six operations

Sobel Kernel
function sobel!(mag, F)
r = (blockIdx().x - Int32(1)) * blockDim().x + threadIdx().x
c = (blockIdx().y - Int32(1)) * blockDim().y + threadIdx().y
H, W = size(F)
if r <= H && c <= W
rm, rp = clamp(r - 1, 1, H), clamp(r + 1, 1, H)  # clamp
cm, cp = clamp(c - 1, 1, W), clamp(c + 1, 1, W)
@inbounds begin
gx = (F[rm, cp] + 2f0 * F[r, cp] + F[rp, cp]) -
(F[rm, cm] + 2f0 * F[r, cm] + F[rp, cm])
gy = (F[rp, cm] + 2f0 * F[rp, c] + F[rp, cp]) -
(F[rm, cm] + 2f0 * F[rm, c] + F[rm, cp])
mag[r, c] = sqrt(gx * gx + gy * gy)
end
end
return nothing
end
Eight reads and ten multiply-adds per pixel, the weights in the code. A
shared-memory tile cuts the reads to 1.3 per pixel.

Edges in Security Work
Perimeter lines: a fence, a door frame or a road edge is a long
straight edge; a change of the edge map along it is an intrusion cue

Tamper detection: a defocused or covered camera loses its

Failure Modes
Lighting change: a cloud or a lamp shifts every pixel at once and
the whole frame becomes foreground; a gain correction from the
intensity histogram or a temporarily larger α absorbs it
Repetitive motion: swaying branches, rain, a flag: motion that is
background; a per-pixel variance or a mixture of Gaussians per
pixel replaces the single mean
Camera shake: every edge moves by a pixel and every edge
becomes foreground; the blur helps, alignment of the frame to the
background cures it
The cost of a false alarm: an operator who sees a hundred false
alarms a day stops looking at them; the same arithmetic as the
false-positive rate of lecture 12

The Pipeline as a Function
function process_frame!(s, f)  # one frame: host bytes to result
copyto!(s.raw, f)  # upload, 2 MB
s.x .= Float32.(s.raw)
@cuda threads=(16, 16) blocks=bl blur_rows!(s.tmp, s.x, s.u)
@cuda threads=(16, 16) blocks=bl blur_cols!(s.b, s.tmp, s.u)
@cuda threads=(16, 16) blocks=bl sobel!(s.mag, s.b)
s.mask .= (abs.(s.b .- s.prev) .> s.τ) .& s.band
s.prev, s.b = s.b, s.prev
return count(s.mask), Array(any(s.mask, dims=2)), Array(any(s.mask, dims=1))
end
function run_clip!(s, frames)  # the whole clip, state reset
blur!(s.prev, s, frames[1])  # upload, convert, blur frame 1
for f in frames[2:end]
process_frame!(s, f)  # the flags wait for the result
end
end
The output boundary is the count and the 3 000 flags on the host; everything before it, the upload
included, is inside the function that is timed.
